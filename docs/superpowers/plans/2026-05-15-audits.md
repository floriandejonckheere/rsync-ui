# Audits Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the Audits feature, which logs every SSH command executed on user servers, with a filterable UI and a show page for command/output detail.

**Architecture:** A new `audits` table captures each `ssh.exec!` call made through `Servers::SSHService`, storing the command, output, exit status, and optional user/job/job_run context. The `SSHService` base class is modified to capture exit codes via Net::SSH channels and create `Audit` records when user context is available. A read-only controller (index/show) with ActionPolicy authorization and a feature flag guard provides the UI.

**Tech Stack:** Rails 8.0, PostgreSQL, ActionPolicy, Pagy, Hotwire/Turbo, Tailwind CSS 4.x, I18n, RSpec/FactoryBot, Net::SSH channels

```ruby
class Audit < ApplicationRecord
  belongs_to :server
  belongs_to :user, optional: true
  belongs_to :job, optional: true
  belongs_to :job_run, optional: true

  validates :command,     presence: true
  validates :executed_at, presence: true

  scope :by_server,  ->(server_id) { where(server_id:)  if server_id.present? }
  scope :by_user,    ->(user_id) { where(user_id:)    if user_id.present? }
  scope :by_command, ->(command) { where("command ILIKE ?", "%#{sanitize_sql_like(command)}%") if command.present? }

  scope :executed_from, ->(from) { where(executed_at: from..) if from.present? }
  scope :executed_to,   ->(to) { where(executed_at: ..to)   if to.present? }
end
```

```ruby
class AuditPolicy < ApplicationPolicy
  authorize :user

  scope_for :relation do |relation|
    next relation if user.admin?

    relation.where(user:)
  end

  def index?
    user.present?
  end

  def show?
    user.admin? || record.user == user
  end
end
```

Create `app/controllers/audits_controller.rb`:

```ruby
# frozen_string_literal: true

class AuditsController < ApplicationController
  include Filterable
  include Sortable

  before_action :authenticate_user!
  before_action :ensure_audits_enabled
  before_action :set_audit, only: [:show]

  def index
    @servers = authorized_scope(Server.order(:name), type: :relation)
    @users = authorized_scope(User.order(:first_name, :last_name), type: :relation)

    scope = authorized_scope(Audit.includes(:server, :user, :job, :job_run).all, type: :relation)
    scope = scope.by_server(@filters[:server_id])
    scope = scope.by_user(@filters[:user_id])
    scope = scope.by_command(@filters[:command])
    scope = scope.executed_from(parse_datetime(@filters[:executed_at_from]))
    scope = scope.executed_to(parse_datetime(@filters[:executed_at_to]))
    scope = sort_for(scope, allowed: ["executed_at"], default: { executed_at: :desc })

    @pagy, @audits = pagy(scope)

    authorize! :audit
  end

  def show
    authorize! @audit
  end

  private

  def set_audit
    @audit = Audit.find(params[:id])
  end

  def ensure_audits_enabled
    raise ActionController::RoutingError, "Not Found" unless Configuration.get("audits")
  end

  def filter_params
    params
      .fetch(:filter, {})
      .permit(
        :server_id,
        :user_id,
        :command,
        :executed_at_from,
        :executed_at_to,
      )
  end
end
```

```ruby
RSpec.describe Servers::SSHService do
  # SSHService is abstract; test audit side effects via ConnectionService
  # (a concrete subclass) with the Net::SSH stack stubbed out.

  let(:server) { create(:server) }
  let(:user) { create(:user) }

    ssh = instance_double(Net::SSH::Connection::Session)
    channel = instance_double(Net::SSH::Connection::Channel)

    allow(Net::SSH).to receive(:start).and_yield(ssh)
    allow(ssh).to receive(:open_channel).and_yield(channel)
    allow(ssh).to receive(:loop)
    allow(channel).to receive(:exec).and_yield(channel, true)
    allow(channel).to receive(:on_data) { |&block| block.call(channel, output) }
    allow(channel).to receive(:on_extended_data)

      if name == "exit-status"

        allow(data).to receive(:read_long).and_return(exit_code)
        block.call(nil, data)
```

```ruby
# frozen_string_literal: true

module Servers
  class SSHService < ApplicationService
    CONNECT_TIMEOUT = 10

    attr_reader :server, :user, :job, :job_run

    def initialize(server, user: nil, job: nil, job_run: nil)
      super()

      @server = server
      @user = user
      @job = job
      @job_run = job_run
    end

    def call
      output = String.new
      exit_code = nil
      executed_at = Time.zone.now

      Net::SSH.start(server.host, server.username, ssh_options) do |ssh|
        ssh.open_channel do |channel|
          channel.exec(command) do |_ch, _success|
            channel.on_data { |_, data| output << data }
            channel.on_extended_data { |_, _, data| output << data }
            channel.on_request("exit-status") do |_, data|
              exit_code = data.read_long
            end
          end
        end

        ssh.loop
      end

      create_audit(output, exit_code, executed_at)
      output
    end

    protected

    def command
      raise NotImplementedError
    end

    private

    def create_audit(output, exit_status, executed_at)
      return unless Configuration.get("audits")
      return unless user.present?

      Audit.create!(
        server:,
        user:,
        job:,
        job_run:,
        command:,
        output:,
        exit_status:,
        executed_at:,
      )
    end

    def ssh_options
      opts = {
        port:            server.port,
        timeout:         CONNECT_TIMEOUT,
        non_interactive: true,
        verify_host_key: :never,
      }

      if server.ssh_key.present?
        opts[:key_data] = [server.ssh_key]
        opts[:keys_only] = true
      elsif server.password.present?
        opts[:password] = server.password
        opts[:auth_methods] = ["password"]
      end

      opts
    end
  end
end
```
